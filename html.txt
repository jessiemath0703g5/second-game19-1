    // --- 繪圖輔助函數 (動態產生幾何圖形 SVG) ---
    function drawCuboid(w, h, d, labelW, labelH, labelD, isCut = false) {
        const maxDim = Math.max(w, h, d);
        const scale = 120 / maxDim; 
        const dw = w * scale;
        const dh = h * scale;
        const dd = d * scale * 0.5; 

        const svgW = dw + dd + 80;
        const svgH = dh + dd + 80;
        const startX = 40;
        const startY = dd + 30;

        let cutLines = '';
        if(isCut) {
            const midY = startY + dh/2;
            cutLines = `
            <line x1="${startX}" y1="${midY}" x2="${startX+dw}" y2="${midY}" stroke="red" stroke-width="3" stroke-dasharray="6,4"/>
            <line x1="${startX+dw}" y1="${midY}" x2="${startX+dw+dd}" y2="${midY-dd}" stroke="red" stroke-width="3" stroke-dasharray="6,4"/>
            `;
        }

    function drawStackedCubes(bigA, smallA) {

        const dd2 = dw2 * 0.5;

        const svgW = dw1 + dd1 + 80;
        const svgH = dw1 + dw2 + dd1 + 80;
        const startX1 = 40;
        const startY1 = dd1 + dw2 + 30;

        const startX2 = startX1 + (dw1 - dw2)/2;
        const startY2 = startY1 - dw2;

        return `<svg width="100%" height="240" viewBox="0 0 ${svgW} ${svgH}" style="max-height: 240px; display: block; margin: 15px auto; overflow: visible;">
            <polygon points="${startX1},${startY1} ${startX1+dw1},${startY1} ${startX1+dw1+dd1},${startY1-dd1} ${startX1+dd1},${startY1-dd1}" fill="#E2E8F0" stroke="#475569" stroke-width="2"/>

    // 洗牌函數
    function shuffleArray(array) {
        for (let i = array.length - 1; i > 0; i--) {
            const j = Math.floor(Math.random() * (i + 1));
            [array[i], array[j]] = [array[j], array[i]];
        }
        return array;
    }

    function startGame() {
        if(selectedLevel === 0 || baseTime === 0) {
            alert("請確實選擇【關卡】與【難度】！");
            return;
        }
        
        score = 0;
        combo = 0;
        totalQuestions = 0;
        correctAnswersCount = 0;
        timeRemaining = baseTime;
        isPaused = false;
        roundHistory = [];
        
        updateStats();
        disableWhiteboard();
        showScreen('screen-game');
        
        timerInterval = setInterval(updateTimer, 1000);
        
        // 初始化題庫順序並洗牌
        if (selectedLevel === 1) {
            currentLevelTypes = shuffleArray([...Array(15).keys()].map(i => i + 1)); // 假設有 15 種題型
        } else {
            currentLevelTypes = shuffleArray([...Array(15).keys()].map(i => i + 1)); // 假設有 15 種題型
        }
        currentTypeIndex = 0;

        nextQuestion();
    }

    function restartGame() {
        if(timerInterval) clearInterval(timerInterval);
        startGame();
    }

    function updateTimer() {
        if(isPaused) return;

        updateStats();
        if(timeRemaining <= 0) {
            endGame();

    }

    function getMultiplier() {
        if (combo <= 1) return 1.0;

        if (combo <= 5) return 1.5;
        return 2.0;
    }

    // --- Question Generators ---
    let currentLevelTypes = [];
    let currentTypeIndex = 0;

    function nextQuestion() {
        document.getElementById('answer-input').value = '';
        document.getElementById('answer-input').focus();
        
        let q;
        let maxTries = 100; 
        
        // 確保洗牌的題型都出過一輪
        if (currentTypeIndex >= currentLevelTypes.length) {
             if (selectedLevel === 1) {
                currentLevelTypes = shuffleArray([...Array(15).keys()].map(i => i + 1));
            } else {
                currentLevelTypes = shuffleArray([...Array(15).keys()].map(i => i + 1));
            }
            currentTypeIndex = 0;
        }

        let qType = currentLevelTypes[currentTypeIndex];

        do {
            if (selectedLevel === 1) {
                q = generateLevel1Question(qType);
            } else {
                q = generateLevel2Question(qType);
            }
            maxTries--;
            // 如果產生了重複的題目(以答案和文字為判斷標準)，重新產生參數
            if (previousQuestions.includes(q.ans + q.text) && maxTries > 0) {
                 continue;
            } else {
                break;
            }
        } while (true);
        
        currentTypeIndex++;

        previousQuestions.push(q.ans + q.text);
        if(previousQuestions.length > 50) previousQuestions.shift(); 

        currentQuestion = q;
        document.getElementById('question-text').innerHTML = currentQuestion.text;
    }

    function randInt(min, max) {
        return Math.floor(Math.random() * (max - min + 1)) + min;
    }

    // --- Validation & Feedback ---
    function checkAnswer() {
        if(isPaused) return;
        const inputStr = document.getElementById('answer-input').value.trim();
        if(inputStr === '') return;
        
        const userAns = parseInt(inputStr, 10);
        const correctAns = currentQuestion.ans;
        const isCorrect = (userAns === correctAns);
        
        totalQuestions++;

        // 紀錄作答歷史供復盤使用
        roundHistory.push({
            q: currentQuestion.text,
            uAns: isNaN(userAns) ? inputStr : userAns,
            cAns: correctAns,
            isC: isCorrect
        });

        if(isCorrect) {
            combo++;
            correctAnswersCount++;
            let points = 100 * getMultiplier();
            score += points;
            timeRemaining += 5;
            showTeacherFeedback(true, `太棒了！答對囉！\n+5 秒`);
        } else {
            combo = 0;
            timeRemaining -= 5;
            showTeacherFeedback(false, `差一點點！\n正確答案是 ${correctAns}\n-5 秒`);
        }
        
        updateStats();
        if(timeRemaining > 0) {
            nextQuestion();
        } else {
            endGame();
        }
    }

        const isLeft = Math.random() > 0.5;
        popup.className = isLeft ? 'left' : 'right';
        
        img.src = isLeft ? 'jessie.png' : 'JESSIEMATH-Q.png';
        img.style.width = randInt(200, 300) + 'px';

        const baseMsg = isCorrect ? goodPhrases[randInt(0, goodPhrases.length-1)] : badPhrases[randInt(0, badPhrases.length-1)];
        msg.innerText = `${baseMsg}\n${extraMsg}`;
        msg.style.borderColor = isCorrect ? '#10B981' : '#EF4444'; 

        popup.classList.add('show');
        
        setTimeout(() => {
            popup.classList.remove('show');
        }, 3000);
    }

    // --- Game Over & Leaderboard ---
    function endGame() {
        if(timerInterval) clearInterval(timerInterval);
        isPaused = true;
        disableWhiteboard();
        
        saveScore();
        showSummary(); // 進入復盤頁面
    }

    function showSummary() {
        const wrongCount = totalQuestions - correctAnswersCount;
        document.getElementById('summary-score').innerText = Math.floor(score);
        document.getElementById('summary-correct').innerText = `${correctAnswersCount} / ${totalQuestions}`;
        document.getElementById('summary-wrong').innerText = `${wrongCount} / ${totalQuestions}`;

        const tbody = document.getElementById('summary-body');
        tbody.innerHTML = '';
        
        if (roundHistory.length === 0) {
            tbody.innerHTML = '<tr><td colspan="4">本次無答題紀錄。</td></tr>';
        } else {
            roundHistory.forEach((item, idx) => {
                const color = item.isC ? '#10B981' : '#EF4444';
                tbody.innerHTML += `
                    <tr>
                        <td>${idx + 1}</td>
                        <td class="q-text">${item.q}</td>
                        <td style="color: ${color}; font-weight: bold;">${item.uAns}</td>
                        <td style="color: #10B981; font-weight: bold;">${item.cAns}</td>
                    </tr>
                `;
            });
        }
        showScreen('screen-summary');
    }

    function getDiffName() {
        if(selectedDiff === 'basic') return "基礎";
        if(selectedDiff === 'adv') return "進階";
        return "魔王";
    }

    function saveScore() {
        if (score === 0 && totalQuestions === 0) return; 
        const record = {
            name: playerName,
            level: `第 ${selectedLevel} 關`,
            diff: getDiffName(),
            score: Math.floor(score),
            correct: correctAnswersCount,
            total: totalQuestions,
            date: new Date().toLocaleString()
        };
        
        let lb = JSON.parse(localStorage.getItem('jessieMathLeaderboard')) || [];
        lb.push(record);
        lb.sort((a, b) => b.score - a.score);
        lb = lb.slice(0, 10); 
        
        localStorage.setItem('jessieMathLeaderboard', JSON.stringify(lb));
    }

    function renderLeaderboard() {
        const lb = JSON.parse(localStorage.getItem('jessieMathLeaderboard')) || [];
        const tbody = document.getElementById('leaderboard-body');
        tbody.innerHTML = '';
        
        if(lb.length === 0) {
            tbody.innerHTML = '<tr><td colspan="6">尚無紀錄，趕快成為第一個挑戰者吧！</td></tr>';
            return;
        }

        lb.forEach((row, index) => {
            let rankHtml = '';
            if (index === 0) rankHtml = '<span class="rank-icon">🥇</span>';
            else if (index === 1) rankHtml = '<span class="rank-icon">🥈</span>';
            else if (index === 2) rankHtml = '<span class="rank-icon">🥉</span>';
            else rankHtml = `<span>${index + 1}</span>`;

            tbody.innerHTML += `
                <tr>
                    <td>${rankHtml}</td>
                    <td>${row.name}</td>
                    <td>${row.level}</td>
                    <td>${row.diff}</td>
                    <td>${row.score} (${row.correct}/${row.total})</td>
                    <td style="font-size: 0.85em; color: #64748B;">${row.date}</td>
                </tr>
            `;
        });
    }

    // --- Whiteboard Logic ---
    const canvas = document.getElementById('whiteboard');
    const ctx = canvas.getContext('2d');
    let isDrawing = false;
    let wbEnabled = false;

    function resizeCanvas() {
        canvas.width = window.innerWidth;
        canvas.height = window.innerHeight;
    }
    window.addEventListener('resize', resizeCanvas);

    // 將事件綁定在 document 上，讓滑鼠與觸控事件能穿透（因 container pointer-events 為 none）
    document.addEventListener('mousedown', (e) => {
        if(!wbEnabled) return;
        const target = e.target;
        // 若點擊的是按鈕、連結或輸入框，不要畫畫，讓原本的點擊事件正常觸發
        if(target && (['INPUT', 'BUTTON', 'A'].includes(target.tagName) || target.closest('button') || target.closest('a'))) return;
        startDrawing(e);
    });

    document.addEventListener('mousemove', (e) => {
        if(!wbEnabled || !isDrawing) return;
        draw(e);
    });

    document.addEventListener('mouseup', stopDrawing);

    document.addEventListener('touchstart', (e) => {
        if(!wbEnabled) return;
        const target = e.target;
        if(target && (['INPUT', 'BUTTON', 'A'].includes(target.tagName) || target.closest('button') || target.closest('a'))) return;
        if (e.cancelable) e.preventDefault();
        startDrawing(e.touches[0]);
    }, {passive: false});

    document.addEventListener('touchmove', (e) => {
        if(!wbEnabled || !isDrawing) return;
        if (e.cancelable) e.preventDefault();
        draw(e.touches[0]);
    }, {passive: false});

    document.addEventListener('touchend', stopDrawing);

    function startDrawing(e) {
        if(!wbEnabled) return;
        isDrawing = true;
        ctx.beginPath();
        const rect = canvas.getBoundingClientRect();
        ctx.moveTo(e.clientX - rect.left, e.clientY - rect.top);
    }

    function draw(e) {
        if (!isDrawing || !wbEnabled) return;
        const rect = canvas.getBoundingClientRect();
        ctx.lineTo(e.clientX - rect.left, e.clientY - rect.top);
        ctx.strokeStyle = '#000000'; 
        ctx.lineWidth = 3;
        ctx.lineCap = 'round';
        ctx.stroke();
    }

    function stopDrawing() {
        isDrawing = false;
    }

    function toggleWhiteboard() {
        wbEnabled = !wbEnabled;
        const btn = document.getElementById('wb-toggle-btn');
        const controls = document.getElementById('wb-controls');
        
        if(wbEnabled) {
            resizeCanvas();
            controls.style.display = 'flex';
            btn.innerText = '❌ 關閉白板';
            document.body.style.userSelect = 'none'; // 防止畫畫時反白到文字
            document.body.style.cursor = 'crosshair';
        } else {
            controls.style.display = 'none';
            btn.innerText = '🖍️ 開啟白板';
            document.body.style.userSelect = '';
            document.body.style.cursor = 'default';
            isDrawing = false;
        }
    }

    function disableWhiteboard() {
        wbEnabled = false;
        document.getElementById('wb-controls').style.display = 'none';
        document.getElementById('wb-toggle-btn').innerText = '🖍️ 開啟白板';
        document.body.style.userSelect = '';
        document.body.style.cursor = 'default';
        clearWhiteboard();
    }

    function clearWhiteboard() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
    }
